9. Which cuisines perform well despite having fewer restaurants?
 Business Value: Highlights niche opportunities for differentiation.


# SQL query to find high-rated cuisines with fewer restaurants
query = """
SELECT
    cuisines,
    AVG(rate) AS average_rating,
    COUNT(*) AS restaurant_count
FROM
    cleaned_data
GROUP BY
    cuisines
HAVING
    average_rating >= 4.5 AND restaurant_count > 5 -- High rating (>=4.5) and a minimum count for representativeness
ORDER BY
    restaurant_count ASC, average_rating DESC -- Order by fewer restaurants first, then by higher rating
LIMIT 10; -- Displaying top 10 niche opportunities
"""

11. Which locations are ideal for premium restaurant onboarding?
 Business Value: Combines cost, rating, and location insights to guide premium expansion.


# SQL query to find ideal locations for premium restaurant onboarding
# Filter for 'Expensive' and 'Very Expensive' segments, then rank by average rating and restaurant count
query = """
SELECT
    location,
    AVG(rate) AS average_premium_rating,
    COUNT(*) AS premium_restaurant_count
FROM
    cleaned_data
WHERE
    price_segment IN ('Expensive', 'Very Expensive')
GROUP BY
    location
HAVING
    premium_restaurant_count > 5 -- Ensure a minimum number of premium restaurants for statistical significance
ORDER BY
    average_premium_rating DESC, premium_restaurant_count ASC
LIMIT 10; -- Displaying top 10 ideal locations
"""

12. Which locations show high demand but lower average ratings?
 Business Value: Indicates areas where quality improvement initiatives are needed.

# SQL query to find locations with high demand but lower average ratings
query = """
SELECT
    location,
    COUNT(*) AS restaurant_count,
    AVG(rate) AS average_rating
FROM
    cleaned_data
GROUP BY
    location
HAVING
    restaurant_count > 100 -- Define 'high demand' as more than 100 restaurants
ORDER BY
    average_rating ASC, restaurant_count DESC -- Order by lowest average rating first, then by demand
LIMIT 10; -- Displaying top 10 locations matching the criteria
"""
